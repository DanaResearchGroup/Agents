import pandas as pd
import time
import cantera as ct
import numpy as np

print(f"Running Cantera version: {ct.__version__}")

# Specify the base path for the mechanism files
mech_file = 'path/to/your/mechanism/files/'  # Update this to your actual path

# Define your keys with original names, below is an example relevant for the provided SI file, chem_annotated.yaml
keys = ['HOCHO(1)', 'CO2(15)', 'CO(13)', 'H2(4)']

# Set the maximum simulation time
max_simulation_time = 3  # seconds

try:
    gas = ct.Solution(mech_file)
except ct.CanteraError as e:
    print(f"Error loading mechanism file {mech_file}: {e}")

# Verify and list available species
print(f"Available species:", gas.species_names)

Ts = np.arange(400, 3000, 40)

P = ct.one_atm

# This is relevant for pyrolysis conditions, change the concentrations for oxidation conditions as needed:
fa = 0.02
o2 = 0
Ne = 1 - fa - o2

X = {"HOCHO(1)": fa, "Ne": Ne}  # Ensure species names are correct

# Reactor parameters
residence_time = 2  # s
reactor_volume = 87 * (1e-2) ** 3  # m3

results = {key: [] for key in keys}

# Map keys to species indices
species_indices = {}
for key in keys:
    try:
        species_indices[key] = gas.species_index(key)
    except KeyError:
        print(f"Warning: Species '{key}' not found in the mechanism. Concentration will be set to NaN.")
        species_indices[key] = None  # Assign None to handle missing species

for T in Ts:
    try:
        gas.TPX = T, P, X
    except ct.CanteraError as e:
        print(f"Error setting T, P, X for T={T} K: {e}")
        for key in keys:
            results[key].append(np.nan)
        continue  # Skip to the next temperature

    # Initialize reactors and controllers
    fuel_air_mixture_tank = ct.Reservoir(gas)
    exhaust = ct.Reservoir(gas)
    stirred_reactor = ct.IdealGasReactor(gas, energy="off", volume=reactor_volume)
    mass_flow_controller = ct.MassFlowController(
        upstream=fuel_air_mixture_tank,
        downstream=stirred_reactor,
        mdot=stirred_reactor.mass / residence_time,
    )
    pressure_regulator = ct.PressureController(
        upstream=stirred_reactor, downstream=exhaust, master=mass_flow_controller
    )
    reactor_network = ct.ReactorNet([stirred_reactor])

    # Set solver tolerances
    reactor_network.atol = 1e-16
    reactor_network.rtol = 1e-16

    # Create a SolutionArray to store the data
    time_history = ct.SolutionArray(gas, extra=["t"])

    t = 0
    counter = 1
    t2 = False
    max_retries = 3
    retries = 0

    while t < max_simulation_time and retries < max_retries:
        try:
            t = reactor_network.step()
            if not t2 and t > 1.9999:
                t = 2.0
                t2 = True

            # Store data every 10 steps or at final time
            if counter % 10 == 0 or t == 2.0:
                time_history.append(stirred_reactor.thermo.state, t=t)

            counter += 1

        except ct.CanteraError as e:
            print(f"CanteraError at T={T} K, t={t} s: {e}")
            retries += 1
            print(f"Retrying simulation for T={T} K (Attempt {retries}/{max_retries})...")
            # Relax tolerances for retry
            reactor_network.atol *= 2
            reactor_network.rtol *= 2

    if retries == max_retries:
        print(f"Failed to converge for T={T} K after {max_retries} attempts.")
        # Append NaN for all keys
        for key in keys:
            results[key].append(np.nan)
        continue  # Move to the next temperature

    # Check if any data was recorded
    if len(time_history.t) == 0:
        print(f"No data recorded for T={T} K.")
        for key in keys:
            results[key].append(np.nan)
        continue

    # Extract species concentrations at t = 2.0 s
    for key in keys:
        idx = species_indices.get(key)
        if idx is None:
            # Species not found, append NaN or handle appropriately
            results[key].append(np.nan)
            continue

        try:
            # Find the index where time is closest to 2.0 s
            index = (np.abs(time_history.t - 2.0)).argmin()
            concentration = time_history.X[index, idx]
            results[key].append(concentration)
        except IndexError:
            print(f"Time history does not contain t=2.0 s, T={T} K.")
            results[key].append(np.nan)
